```python
TARGET_RETENTION = 0.60
GG_377_RETENTION = 0.40
SUBJECTS = ("377", "386", "394")
METHODS = ("saga", "gaussian_grouping", "sggs", "a5")


def build_temporal_windows(*, cameras=(21, 22, 23), anchors=(180, 420, 540), radius=10):
    windows = []
    seen = set()
    for camera in cameras:
        for anchor in anchors:
            frames = list(range(int(anchor) - int(radius), int(anchor) + int(radius) + 1))
            if frames[0] < 0:
                raise ValueError("temporal window contains a negative frame")
            keys = {(int(camera), frame) for frame in frames}
            if seen & keys:
                raise ValueError("temporal windows overlap")
            seen.update(keys)
            windows.append({
                "camera": int(camera),
                "anchor": int(anchor),
                "window": f"c{int(camera):02d}_a{int(anchor):06d}",
                "frames": frames,
            })
    return windows
```

`resolve_frozen_operating_point` 使用 `baseline`、`retention` 唯一匹配。A5 匹配 `A5`，三个外部方法匹配 `B4`；GG-377 只接受 0.40，其他项只接受 0.60。
